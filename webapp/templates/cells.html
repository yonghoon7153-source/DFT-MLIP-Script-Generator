cp    data/raw/                                      # 원 데이터 (커밋되지 않는다)
python3 tools/cells/import_cell.py --propose data/raw/   # 파일명에서 제안값 보기
#  → data/registry/cells.csv 에 행을 적는다 (cell_id, formation_cutoff_v_li, active_mass_mg, …, raw_file)
python3 tools/cells/import_cell.py --cell                # 사이클 요약 계산 → data/cells//
